Task Scheduling

nano script.sh
#!/bin/bash
echo "Hello from scheduled task"
chmod +x script.sh
pwd
./script.sh
crontab -e
*/10 * * * * /home/ubuntu/script.sh >> /home/ubuntu/output.log 2>&1
sudo systemctl enable --now cron
sudo systemctl status cron
Wait one minute..  cat output.log
