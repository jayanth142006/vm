import socket import threading

clients=[]
def handle_clients(c):
    while true:
        try:
            data=c.recv(1096).decode()
            if not data:
                break
            print(data)
            for client in clients:
                if client!=c:
                    client.send(data.encode())
        except:
            break
    clients.remove(c)
    c.close()
s=socket.socket()
s.bind(("localhost",3000))
s.listen(1)
print("server")
while True:
    c,addr=s.accept()
    clients.append(c)
    print("client")
    threading.Thread(target=handle_clients,args=(c,)).start()
